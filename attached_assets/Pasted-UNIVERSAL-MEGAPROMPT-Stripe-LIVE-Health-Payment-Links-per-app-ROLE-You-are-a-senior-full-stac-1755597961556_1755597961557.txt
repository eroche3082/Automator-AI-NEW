3.1 Node script (Replit-ready) to create or reuse catalog + links

Create stripe_links.ts (or .js) and run with the app’s LIVE secret:

/**
 * Usage:
 *   STRIPE_SECRET_KEY=sk_live_xxx npx ts-node stripe_links.ts --app JETZZ --plans basic=1900/mo pro=4900/mo elite=9900/mo --success https://jetzz.socialbrands.ai/thanks --cancel https://jetzz.socialbrands.ai/cancel
 *
 * Notes:
 * - Idempotent by name: if product/price exists, it reuses it.
 * - Creates/returns Payment Links for each plan.
 */
import Stripe from "stripe";
import crypto from "crypto";

const args = process.argv.join(" ");
const appName = /--app\s+(\S+)/.exec(args)?.[1] || "APP";
const successUrl = /--success\s+(\S+)/.exec(args)?.[1] || "https://socialbrands.ai/success";
const cancelUrl  = /--cancel\s+(\S+)/.exec(args)?.[1] || "https://socialbrands.ai/cancel";
const plansArg   = /--plans\s+([^]+)$/.exec(args)?.[1] || ""; // e.g. "basic=1900/mo pro=4900/mo"
const planDefs   = Array.from(plansArg.matchAll(/(\w+)=([0-9]+)\/(mo|yr)/g));

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, { apiVersion: "2023-10-16" });

async function ensureProduct(name: string) {
  const search = await stripe.products.search({ query: `name:"${name}" AND active:'true'` });
  if (search.data[0]) return search.data[0];
  return stripe.products.create({ name, active: true }, { idempotencyKey: "prod_" + crypto.randomUUID() });
}

async function ensurePrice(product: string, unit: number, interval: "month"|"year") {
  const list = await stripe.prices.list({ product, active: true, limit: 100 });
  const found = list.data.find(p => p.unit_amount === unit && p.recurring?.interval === interval);
  if (found) return found;
  return stripe.prices.create({
    product, currency: "usd", unit_amount: unit, recurring: { interval }
  }, { idempotencyKey: "price_" + crypto.randomUUID() });
}

async function ensurePaymentLink(priceId: string) {
  // Reuse if already created with same price (best-effort: fetch latest link with that price)
  const links = await stripe.paymentLinks.list({ limit: 100 });
  const existing = links.data.find(l => l.line_items?.some(li => (li.price as any)?.id === priceId));
  if (existing) return existing;
  return stripe.paymentLinks.create({
    line_items: [{ price: priceId, quantity: 1 }],
    after_completion: { type: "redirect", redirect: { url: successUrl } },
    metadata: { app: appName }
  }, { idempotencyKey: "plink_" + crypto.randomUUID() });
}

(async () => {
  console.log(`== Stripe LIVE Payment Links for ${appName} ==`);
  for (const m of planDefs) {
    const plan = m[1]; const cents = parseInt(m[2], 10); const per = m[3] === "mo" ? "month" : "year";
    const productName = `${appName} — ${plan.toUpperCase()} Subscription`;
    const product = await ensureProduct(productName);
    const price   = await ensurePrice(product.id, cents, per as any);
    const plink   = await ensurePaymentLink(price.id);
    console.log(`[${plan}] product=${product.id} price=${price.id} link=${plink.url}`);
  }
})();
