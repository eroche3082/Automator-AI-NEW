Instruction: Create the file interactive-qc.spec.ts with the following content:

TypeScript

// File: interactive-qc.spec.ts
import { test, expect } from '@playwright/test';

const BASE_URL = 'http://localhost:5000';

test.describe('Clipper.AI Interactive Frontend QC', () => {

  test('BUG HUNT: Dashboard - Video Play Button Functionality', async ({ page }) => {
    await page.goto(`${BASE_URL}/dashboard`);
    
    // Wait for video cards to be visible
    await expect(page.locator('.video-card').first()).toBeVisible({ timeout: 10000 });

    const playButton = page.locator('.video-card .play-button').first();
    await expect(playButton).toBeVisible();
    await playButton.click();

    // Give the video a moment to start playing
    await page.waitForTimeout(1500);

    // Verify the video is actually playing by checking its currentTime property
    const videoTime = await page.evaluate(() => {
      const video = document.querySelector('video');
      return video ? video.currentTime : 0;
    });

    expect(videoTime).toBeGreaterThan(0);
  });

  test('BUG HUNT: Workspace Management - Check for NaN Error', async ({ page }) => {
    await page.goto(`${BASE_URL}/workspace-management`); // Adjust URL if different

    // Find the element that displays the member list or count
    const memberElement = page.locator('.member-list-container'); // Adjust selector as needed
    await expect(memberElement).toBeVisible({ timeout: 10000 });

    const elementText = await memberElement.innerText();
    
    // Assert that the text does NOT contain "NaN"
    expect(elementText).not.toContain('NaN');
  });
  
  test('BUG HUNT: Analytics - Date Filter Functionality', async ({ page }) => {
    await page.goto(`${BASE_URL}/analytics-dashboard`); // Adjust URL if different

    const chart = page.locator('.metrics-chart');
    await expect(chart).toBeVisible({ timeout: 10000 });

    // Promise to wait for the API call that refreshes chart data
    const responsePromise = page.waitForResponse(resp => resp.url().includes('/api/analytics/data')); // Adjust API endpoint as needed

    // Click the date filter button to open the calendar/selector
    await page.click('.date-filter-button'); // Adjust selector
    
    // Click a new date range, e.g., "Last 7 Days"
    await page.click('.date-range-option[data-value="7days"]'); // Adjust selector

    // Wait for the corresponding API call to complete
    await responsePromise;

    // A simple check to see if the chart might have updated. A more robust test would check chart data.
    await expect(chart).toBeVisible();
    console.log('Analytics filter test completed, API call for data refresh was detected.');
  });
});
